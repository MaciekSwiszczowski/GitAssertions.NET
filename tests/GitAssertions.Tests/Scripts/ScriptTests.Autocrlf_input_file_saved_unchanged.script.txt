# With core.autocrlf=input a CRLF file is committed, then saved again with the same bytes. Git sees the new write time,
# reads the file again, turns its CRLF back into LF, and finds the committed content. Nothing is modified.
git init
git config core.autocrlf input
echo line windows.txt
unix2dos windows.txt
git add windows.txt
git commit -m "Add windows.txt"
unix2dos windows.txt
